### Task 1: Calcul pur — `data/cbprem.ts`

**Files:**
- Create: `apps/web/src/data/cbprem.ts`
- Test: `apps/web/src/data/cbprem.test.ts`

**Interfaces (Produces):**
```ts
export interface PointPremium { t: number; premiumPct: number; }
export function serieCbprem(klinesCb: { t: number; close: number }[], klinesBn: { t: number; close: number }[]): PointPremium[];
// alignement par openTime exact (Map) ; points non appariés OMIS ; premium = (cb − bn)/bn × 100 ;
// closes non finis ou bn <= 0 → point omis ; sortie triée chrono.
export function statsPremium(serie: PointPremium[]): { courant: number | null; moyenne7j: number | null; z30j: number | null };
// courant = dernier ; moyenne7j = moyenne des points des 7 derniers jours (t relatif au dernier point) ;
// z30j = z-score du courant vs toute la série (stdev population, null si < 30 points ou stdev 0).
```
