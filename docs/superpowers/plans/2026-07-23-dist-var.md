```ts
  pct: { p1: number; p5: number; p50: number; p95: number; p99: number };     // en % vs dernierClose

export function distVar(closes: readonly number[], horizons?: readonly number[]): NiveauxVar[] | null;
// null si < 300 closes finis ; rendements log 1-bougie ; par horizon h : sommes GLISSANTES de h rendements
// (fenêtres chevauchantes) ; quantiles par interpolation linéaire ; non-finis écartés en amont.
```
